CREATE DATABASE "supermarket_pos"
    WITH
    OWNER = postgres
    ENCODING = 'UTF8'
    LC_COLLATE = 'Spanish_Honduras.1252'
    LC_CTYPE = 'Spanish_Honduras.1252'
    LOCALE_PROVIDER = 'libc'
    TABLESPACE = pg_default
    CONNECTION LIMIT = -1
    IS_TEMPLATE = False;

2

-- 1. Crear Tabla de Productos
CREATE TABLE IF NOT EXISTS productos (
    id SERIAL PRIMARY KEY,
    codigo VARCHAR(50) UNIQUE NOT NULL, -- Código de barras o ID del PDF
    nombre VARCHAR(255) NOT NULL,
    precio DECIMAL(10, 2) NOT NULL,
    stock_actual INT DEFAULT 100,       -- Iniciamos con 100 como pediste
    stock_minimo INT DEFAULT 10         -- Alerta cuando baje de 10
);

-- 2. Crear Tabla de Ventas (Cabecera)
CREATE TABLE IF NOT EXISTS ventas (
    id SERIAL PRIMARY KEY,
    fecha_venta TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total DECIMAL(10, 2) DEFAULT 0
);

-- 3. Crear Tabla de Detalle de Venta (Items individuales)
CREATE TABLE IF NOT EXISTS detalle_ventas (
    id SERIAL PRIMARY KEY,
    venta_id INT REFERENCES ventas(id),
    producto_id INT REFERENCES productos(id),
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(10, 2) NOT NULL,
    subtotal DECIMAL(10, 2) GENERATED ALWAYS AS (cantidad * precio_unitario) STORED
);

-- 4. Índice para búsquedas rápidas por código de barras
CREATE INDEX idx_codigo_producto ON productos(codigo);
